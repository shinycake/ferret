# Copy to .github/workflows/release.yml (needs a token with the `workflow` scope to push).
name: Release
on:
  push:
    tags: ["v*"]
jobs:
  build:
    runs-on: macos-latest
    timeout-minutes: 60
    steps:
      - uses: actions/checkout@v4
      - run: brew install xcodegen
      - uses: dtolnay/rust-toolchain@stable
        with:
          targets: aarch64-apple-darwin,x86_64-apple-darwin
      - run: scripts/build-fsearch.sh
      - run: |
          xcodegen generate
          xcodebuild -project Ferret.xcodeproj -scheme Ferret -configuration Release -destination 'platform=macOS' -derivedDataPath build/dd CODE_SIGNING_ALLOWED=NO build
      - run: SKIP_CI_PROOF=1 scripts/package.sh
      - uses: actions/upload-artifact@v4
        with:
          name: release
          path: dist/
  notarize:
    needs: build
    if: ${{ secrets.APPLE_ID != '' }}
    runs-on: macos-latest
    steps:
      - run: echo "see docs/NOTARIZATION.md"
  publish:
    needs: build
    runs-on: ubuntu-latest
    permissions:
      contents: write
    steps:
      - uses: actions/download-artifact@v4
        with:
          name: release
          path: dist
      - run: gh release create "$GITHUB_REF_NAME" dist/* --prerelease --repo "$GITHUB_REPOSITORY" --title "$GITHUB_REF_NAME"
        env:
          GH_TOKEN: ${{ github.token }}
